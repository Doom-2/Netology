cmake_minimum_required(VERSION 3.22.0)
project(RacingSimulator CXX)
 
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
 
# exe и библиотека собираются в одну папку, чтобы exe находил DLL рядом с собой
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
 
# Все исходники в UTF-8 (раньше часть файлов была в cp1251, часть в UTF-8)
if(MSVC)
    add_compile_options(/utf-8)
endif()
 
add_subdirectory(RacingSimulatorDLL)
 
add_executable(RacingSimulator RacingSimulator.cpp)
target_link_libraries(RacingSimulator PRIVATE RacingSimulatorDLL)
